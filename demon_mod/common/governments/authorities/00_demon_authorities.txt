# Custom Government Authority for the Demon Mod

	ruler_title = "High Priest"
	ruler_title_female = "High Priestess"

	# Can't use this authority if you are egalitarian

	# AI will prefer this authority if they have the Satanic ethic
	ai_weight = {

		modifier = {
			factor = 2.0
			has_ethic = "ethic_satan"
		}
		modifier = {
			factor = 3.0
			has_ethic = "ethic_fanatic_satan"
		}
	}
}
