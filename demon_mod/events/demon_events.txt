# Events for the Demon Species

namespace = demon

# Event that applies the social outcast opinion modifier at game start
country_event = {
	id = demon.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		# This event should only fire for countries that have the demon trait
		owner_main_species = {
			has_trait = trait_demon_social_outcast
		}
	}

	immediate = {
		# Apply the opinion modifier to every other country
		every_other_country = {
			limit = {
				is_ai = yes # Only apply to AI empires to avoid player frustration
			}
			add_opinion_modifier = {
				who = ROOT
				modifier = demon_species_opinion
			}
		}
	}
}
