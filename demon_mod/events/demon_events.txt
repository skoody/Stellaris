namespace = demon

# Event to grant influence when a pop is purged
country_event = {
	id = demon.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_civic = "civic_satanism" }
			add_resource = { influence = 0.5 }
		}
	}
}

# Event to apply opinion malus at game start
country_event = {
	id = demon.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				is_ai = no
				founder_species = { has_trait = "trait_demon_social_outcast" }
			}
			every_country = {
				limit = {
					is_country_type = default
					NOT = { this = ROOT }
				}
				add_opinion_modifier = {
					who = ROOT
					modifier = "demon_species_opinion"
				}
			}
		}
	}
}
