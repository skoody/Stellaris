namespace = demon

# Event to grant influence when a pop is purged
country_event = {
	id = demon.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_civic = "civic_satanism"
	}

	immediate = {
		add_resource = {
			influence = 0.5
		}
	}
}

# Event to apply opinion malus at game start
country_event = {
	id = demon.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		founder_species = {
			has_trait = "trait_demon_social_outcast"
		}
	}

	immediate = {
		every_country = {
			limit = {
				NOT = { this = ROOT }
			}
			add_opinion_modifier = {
				who = ROOT
				modifier = "demon_species_opinion"
			}
		}
	}
}
